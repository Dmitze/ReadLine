# 📋 Фінальний детальний промпт для Задачі 4: Система заявок на фізичні книги (v3 - ФІНАЛ)

---

## 🎯 Мета задачі

Додати функціонал заявок на отримання фізичних книг. Користувачі зможуть залишати заявки на книги, вказуючи свої контактні дані, а адміністратор зможе переглядати ці заявки та керувати ними. **Книги видаються БЕЗ обмеження терміну** - користувач повертає коли захоче. **Без зазначення конкретного місця в бібліотеці.**

---

## 🔍 Що саме реалізуємо

### Основні компоненти:

1. **База даних** - нові таблиці для заявок
2. **Логіка додавання книг** - прапорець "Є фізично"
3. **Картка книги** - відображення статусу та кнопка заявки
4. **Scene для створення заявки** - wizard збору даних
5. **Список заявок користувача** - перегляд своїх заявок
6. **Адмін-панель** - управління заявками
7. **Система сповіщень** - повідомлення про статуси

```sql
-- Міграція: Створення таблиці заявок на книги
-- Дата: 2025-11-16

-- Таблиця заявок на фізичні книги
CREATE TABLE IF NOT EXISTS book_requests (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  book_id INTEGER NOT NULL,
  user_id INTEGER NOT NULL,
  full_name TEXT NOT NULL,
  callsign TEXT NOT NULL,
  unit TEXT NOT NULL,
  phone TEXT NOT NULL,
  status TEXT NOT NULL DEFAULT 'pending' CHECK(status IN ('pending', 'approved', 'issued', 'returned', 'rejected', 'cancelled')),
  comment TEXT,
  admin_comment TEXT,
  issued_at DATETIME,
  returned_at DATETIME,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  processed_by INTEGER,
  processed_at DATETIME,
  FOREIGN KEY (book_id) REFERENCES books(id) ON DELETE CASCADE
);

-- Індекси для швидкого пошуку
CREATE INDEX IF NOT EXISTS idx_book_requests_book_id ON book_requests(book_id);
CREATE INDEX IF NOT EXISTS idx_book_requests_user_id ON book_requests(user_id);
CREATE INDEX IF NOT EXISTS idx_book_requests_status ON book_requests(status);
CREATE INDEX IF NOT EXISTS idx_book_requests_created_at ON book_requests(created_at);

-- Індекс для пошуку активних заявок користувача на конкретну книгу
CREATE INDEX IF NOT EXISTS idx_book_requests_user_book_active 
ON book_requests(user_id, book_id, status);
```

```sql
-- Міграція: Додавання полів для фізичних примірників книг
-- Дата: 2025-11-16

-- Додаємо поле "чи є фізичний примірник"
ALTER TABLE books ADD COLUMN has_physical_copy BOOLEAN DEFAULT 0;

-- Додаємо поля для управління кількістю
ALTER TABLE books ADD COLUMN physical_copies_total INTEGER DEFAULT 0;
ALTER TABLE books ADD COLUMN physical_copies_available INTEGER DEFAULT 0;

-- Індекс для швидкого пошуку книг з фізичними примірниками
CREATE INDEX IF NOT EXISTS idx_books_has_physical ON books(has_physical_copy);
CREATE INDEX IF NOT EXISTS idx_books_physical_available ON books(physical_copies_available);

-- Оновлюємо існуючі книги
-- Якщо є file_type = 'physical', встановлюємо has_physical_copy = 1
UPDATE books 
SET has_physical_copy = 1, 
    physical_copies_total = 1, 
    physical_copies_available = 1 
WHERE file_type = 'physical';
```

**Детальна реалізація кожного кроку:**

```typescript
// ENTER - Попередження
bookRequestScene.enter(async (ctx) => {
  const bookId = (ctx.scene.state as any).bookId;
  
  if (!bookId) {
    await ctx.reply('❌ Помилка: не вказано книгу');
    return ctx.scene.leave();
  }
  
  const book = await getBookById(bookId);
  if (!book) {
    await ctx.reply('❌ Книгу не знайдено');
    return ctx.scene.leave();
  }
  
  // Перевірити чи є фізична копія
  if (!book.has_physical_copy) {
    await ctx.reply('❌ Для цієї книги неможливо залишити заявку - немає фізичного примірника');
    return ctx.scene.leave();
  }
  
  // Перевірити чи користувач вже має активну заявку

bookRequestScene.on('text', async (ctx) => {
  const state = ctx.scene.state as any;
  const text = ctx.message.text.trim();
  
  // Скасування
  if (text === '❌ Скасувати') {
    await ctx.reply('❌ Створення заявки скасовано.', {
      reply_markup: { remove_keyboard: true }
    });
    return ctx.scene.leave();
  }
  
  // Крок 1: ПІБ
  if (state.step === 'full_name') {
    if (text.length < 5) {
      await ctx.reply('❌ ПІБ занадто короткий. Мінімум 5 символів. Спробуйте ще раз:');
      return;
    }
    
    // Перевірка на кирилицю (опціонально)
    if (!/[а-яА-ЯіІїЇєЄґҐ]/.test(text)) {
      await ctx.reply('❌ Будь ласка, введіть ПІБ кирилицею. Спробуйте ще раз:');
      return;
    }
    
    state.full_name = text;
    state.step = 'callsign';
    
    await ctx.reply(
      '✅ ПІБ збережено!\n\n' +
      '🎯 <b>КРОК 2/4: ПОЗИВНИЙ</b>\n\n' +
      'Введіть ваш позивний:\n\n' +
      '💡 <i>Приклад: Воїн, Сокіл, Грім тощо</i>',
      { parse_mode: 'HTML' }
    );
    return;
  }
  
  // Крок 2: Позивний
  if (state.step === 'callsign') {
    if (text.length < 2) {
      await ctx.reply('❌ Позивний занадто короткий. Мінімум 2 символи. Спробуйте ще раз:');
      return;
    }
    
    state.callsign = text;
    state.step = 'unit';
    
    await ctx.reply(
      '✅ Позивний збережено!\n\n' +
      '🏢 <b>КРОК 3/4: ПІДРОЗДІЛ</b>\n\n' +
      'Введіть ваш підрозділ:\n\n' +
      '💡 <i>Приклад: 1-ша рота, 2-й батальйон, штаб тощо</i>',
      { parse_mode: 'HTML' }
    );
    return;
  }
  
  // Крок 3: Підрозділ
  if (state.step === 'unit') {
    if (text.length < 3) {
      await ctx.reply('❌ Назва підрозділу занадто коротка. Мінімум 3 символи. Спробуйте ще раз:');
      return;
    }
    
    state.unit = text;
    state.step = 'phone';
    
    await ctx.reply(
      '✅ Підрозділ збережено!\n\n' +
      '📞 <b>КРОК 4/4: НОМЕР ТЕЛЕФОНУ</b>\n\n' +
      'Введіть номер телефону:\n\n' +
      '💡 <i>Приклад: +380991234567</i>',
      {
        parse_mode: 'HTML',
        reply_markup: Markup.keyboard([
          [Markup.button.contactRequest('📱 Поділитися номером')],
          ['❌ Скасувати']
        ]).resize().reply_markup
      }
    );
    return;
  }
  
  // Крок 4: Номер телефону (текстом)
  if (state.step === 'phone') {
    // Базова валідація номера
    const phoneRegex = /^(\+380|380|0)\d{9}$/;
    const cleanPhone = text.replace(/[\s\-\(\)]/g, '');
    
    if (!phoneRegex.test(cleanPhone)) {
      await ctx.reply(
        '❌ Невірний формат номера телефону.\n\n' +
        'Введіть номер у форматі:\n' +
        '+380991234567 або 0991234567'
      );
      return;
    }
    
    // Нормалізація номера
    let normalizedPhone = cleanPhone;
    if (normalizedPhone.startsWith('380')) {
      normalizedPhone = '+' + normalizedPhone;
    } else if (normalizedPhone.startsWith('0')) {
      normalizedPhone = '+38' + normalizedPhone;
    }
    
    state.phone = normalizedPhone;
    state.step = 'comment_choice';
    
    await ctx.reply(
      '✅ Номер телефону збережено!\n\n' +
      '💬 Бажаєте додати коментар до заявки?\n\n' +
      '💡 <i>Наприклад: "Заберу завтра після обіду" або "Можна зв\'язатися вечером"</i>',
      {
        parse_mode: 'HTML',
        reply_markup: Markup.inlineKeyboard([
          [Markup.button.callback('✍️ Додати коментар', 'request_add_comment')],
          [Markup.button.callback('⏭️ Пропустити', 'request_skip_comment')]
        ]).reply_markup
      }
    );
    return;
  }
  
  // Крок 5: Коментар (якщо користувач обрав додати)
  if (state.step === 'comment') {
    if (text.length > 500) {
      await ctx.reply('❌ Коментар занадто довгий. Максимум 500 символів.');
      return;
    }
    
    state.comment = text;
    await showConfirmation(ctx);
    return;
  }
});
```

```typescript
export function getDaysAgo(date: string): number {
  const then = new Date(date);
  const now = new Date();
  const diffTime = Math.abs(now.getTime() - then.getTime());
  const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
  return diffDays;
}

export function formatDate(date: string): string {
  return new Date(date).toLocaleDateString('uk-UA', {
    day: '2-digit',
    month: '2-digit',
    year: 'numeric',
    hour: '2-digit',
    minute: '2-digit'
  });
}
```

### Крок 10.2: SQL запити для перевірки

```sql
-- Перевірити чи всі книги мають правильні поля
SELECT id, title, has_physical_copy, physical_copies_total, physical_copies_available 
FROM books;

-- Перевірити заявки
SELECT r.id, r.status, r.user_id, b.title, r.created_at
FROM book_requests r
JOIN books b ON r.book_id = b.id
ORDER BY r.created_at DESC;

-- Перевірити консистентність даних
SELECT 
  b.id,
  b.title,
  b.physical_copies_total,
  b.physical_copies_available,
  (SELECT COUNT(*) FROM book_requests WHERE book_id = b.id AND status IN ('approved', 'issued')) as issued_count
FROM books b
WHERE b.has_physical_copy = 1;

-- Має бути: physical_copies_available + issued_count = physical_copies_total
```

```markdown
# Система заявок на фізичні книги

## Огляд

Система дозволяє користувачам залишати заявки на отримання фізичних примірників книг з бібліотеки. Адміністратор може схвалювати або відхиляти заявки.

## Основні компоненти

### База даних

- **books** - таблиця книг (оновлена):
  - `has_physical_copy` - чи є фізичний примірник (boolean)
  - `physical_copies_total` - загальна кількість примірників
  - `physical_copies_available` - доступна кількість

- **book_requests** - таблиця заявок:
  - Контактні дані користувача
  - Статус заявки
  - Дати створення/видачі/повернення

### Статуси заявок

1. **pending** - очікує розгляду адміном
2. **approved** - схвалено, готова до видачі
3. **issued** - видано користувачу
4. **returned** - повернуто
5. **rejected** - відхилено адміном
6. **cancelled** - скасовано користувачем

### Workflow

```
Користувач створює заявку
        ↓
    [pending]
        ↓
    Адмін розглядає
        ↓
   [approved] або [rejected]
        ↓
Користувач підтверджує отримання
        ↓
    [issued]
        ↓
Користувач повертає книгу
        ↓
    [returned]
```

### Важливі функції

#### bookRequests.ts
- `createBookRequest()` - створити заявку
- `getUserRequests()` - отримати заявки користувача
- `updateRequestStatus()` - оновити статус
- `getRequestsStatistics()` - статистика

#### books.ts
- `decrementAvailableCopies()` - зменшити кількість
- `incrementAvailableCopies()` - збільшити кількість
- `getBookAvailability()` - перевірити наявність

## Обмеження

- Термін користування: **БЕЗ ОБМЕЖЕНЬ**
- Один користувач може мати лише одну активну заявку на книгу
- Не зазначається конкретне місце в бібліотеці

## Безпека

- Валідація всіх введених даних
- Перевірка наявності перед створенням заявки
- Транзакції для критичних операцій (race conditions)
```

```typescript
import { z } from 'zod';

// Схема валідації для створення заявки
export const createRequestSchema = z.object({
  book_id: z.number().int().positive(),
  user_id: z.number().int().positive(),
  full_name: z.string()
    .min(5, 'ПІБ занадто короткий')
    .max(100, 'ПІБ занадто довгий')
    .regex(/[а-яА-ЯіІїЇєЄґҐ]/, 'ПІБ має містити кирилицю'),
  callsign: z.string()
    .min(2, 'Позивний занадто короткий')
    .max(50, 'Позивний занадто довгий'),
  unit: z.string()
    .min(3, 'Назва підрозділу занадто коротка')
    .max(100, 'Назва підрозділу занадто довга'),
  phone: z.string()
    .regex(/^\+380\d{9}$/, 'Невірний формат номера телефону'),
  comment: z.string()
    .max(500, 'Коментар занадто довгий')
    .optional(),
});

// Функція валідації
export function validateCreateRequest(data: unknown) {
  return createRequestSchema.safeParse(data);
}

// Нормалізація номера телефону
export function normalizePhone(phone: string): string {
  const cleaned = phone.replace(/[\s\-\(\)]/g, '');
  
  if (cleaned.startsWith('+380')) {
    return cleaned;
  }
  if (cleaned.startsWith('380')) {
    return '+' + cleaned;
  }
  if (cleaned.startsWith('0')) {
    return '+38' + cleaned;
  }
  
  throw new Error('Невірний формат номера');
}
```

### Крок 12.2: Обробка race conditions

```typescript
// У функції createBookRequest() використовувати транзакцію:

async function createBookRequest(data: BookRequest): Promise<number> {
  return new Promise((resolve, reject) => {
    db.serialize(() => {
      db.run('BEGIN TRANSACTION');
      
      try {
        // 1. Перевірити наявність
        db.get(
          'SELECT physical_copies_available FROM books WHERE id = ?',
          [data.book_id],
          (err, row: { physical_copies_available: number }) => {
            if (err) {
              db.run('ROLLBACK');
              return reject(err);
            }
            
            if (!row || row.physical_copies_available === 0) {
              db.run('ROLLBACK');
              return reject(new Error('Немає доступних примірників'));
            }
            
            // 2. Створити заявку
            db.run(
              'INSERT INTO book_requests (book_id, user_id, full_name, callsign, unit, phone, comment, status) VALUES (?, ?, ?, ?, ?, ?, ?, ?)',
              [data.book_id, data.user_id, data.full_name, data.callsign, data.unit, data.phone, data.comment, 'pending'],
              function (err) {
                if (err) {
                  db.run('ROLLBACK');
                  return reject(err);
                }
                
                const requestId = this.lastID;
                
                // 3. Зменшити кількість доступних
                db.run(
                  'UPDATE books SET physical_copies_available = physical_copies_available - 1 WHERE id = ?',
                  [data.book_id],
                  (err) => {
                    if (err) {
                      db.run('ROLLBACK');
                      return reject(err);
                    }
                    
                    db.run('COMMIT');
                    resolve(requestId);
                  }
                );
              }
            );
          }
        );
      } catch (error) {
        db.run('ROLLBACK');
        reject(error);
      }
    });
  });
}
```
